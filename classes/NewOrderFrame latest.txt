import javax.swing.*;
import javax.swing.border.Border;
import javax.swing.event.DocumentEvent;
import javax.swing.event.DocumentListener;
import javax.swing.text.AbstractDocument;
import javax.swing.text.AttributeSet;
import javax.swing.text.BadLocationException;
import javax.swing.text.DocumentFilter;
import java.awt.*;
import java.awt.event.ActionListener;
import java.awt.event.WindowAdapter;
import java.awt.event.WindowEvent;
import java.text.SimpleDateFormat;
import java.util.ArrayList;
import java.util.Date;
import java.util.List;
import java.util.Locale;
import com.toedter.calendar.JDateChooser;

public class NewOrderFrame extends JFrame {

    private JTextField txtName, txtPhone, txtLocation;
    private JTextField txtOrder, txtTime;
    private JTextField txtPrice, txtAdvance, txtDue;
    private JDateChooser dateChooser;
    private JComboBox<String> cmbOrderType, cmbPaymentMethod, cmbPaymentStatus, cmbAmPm;
    private JButton btnSave, btnCalculate, btnClear, btnBack;

    private Border defaultBorder;
    private final Border errorBorder = BorderFactory.createLineBorder(new Color(220, 53, 69), 2);

    // --- Catering Components & Managers ---
    private CateringPriceManager priceManager;
    private CateringOrderCalculator orderCalculator;

    private JComboBox<String> riceDishCombo;
    private JSpinner riceMunSpinner;

    // Unified Protein Section Components
    private JRadioButton rbBeef, rbChicken, rbFish, rbMutton;
    private ButtonGroup proteinGroup;

    private JLabel lblProteinDish, lblProteinQty;
    private JComboBox<String> proteinDishCombo;
    private JSpinner proteinMunSpinner;

    private JCheckBox proteinCustomerProvidedCheck;

    public NewOrderFrame() {
        this(new CateringPriceManager());
    }

    public NewOrderFrame(CateringPriceManager priceManager) {
        this.priceManager = priceManager;
        this.orderCalculator = new CateringOrderCalculator(priceManager);

        setTitle("New Order - Almayda");
        setSize(650, 850);
        setLocationRelativeTo(null);
        setDefaultCloseOperation(JFrame.DO_NOTHING_ON_CLOSE);
        addWindowListener(new WindowAdapter() {
            @Override
            public void windowClosing(WindowEvent e) {
                confirmExitOrBack();
            }
        });

        JPanel panel = new JPanel();
        panel.setLayout(new GridLayout(0, 2, 10, 10));
        panel.setBorder(BorderFactory.createEmptyBorder(15, 15, 15, 15));

        // Customer Info
        panel.add(new JLabel("Customer Name:"));
        txtName = new JTextField();
        panel.add(txtName);

        panel.add(new JLabel("Phone Number (8 digits starting with 9 or 7):"));
        txtPhone = new JTextField();
        panel.add(txtPhone);

        ((AbstractDocument) txtPhone.getDocument()).setDocumentFilter(new DocumentFilter() {
            @Override
            public void insertString(FilterBypass fb, int offset, String string, AttributeSet attr) throws BadLocationException {
                if (string == null) return;
                if ((fb.getDocument().getLength() + string.length()) <= 8 && (string.isEmpty() || string.matches("\\d+"))) {
                    super.insertString(fb, offset, string, attr);
                }
            }

            @Override
            public void replace(FilterBypass fb, int offset, int length, String text, AttributeSet attrs) throws BadLocationException {
                if (text == null) return;
                int newLength = fb.getDocument().getLength() - length + text.length();
                if (newLength <= 8 && (text.isEmpty() || text.matches("\\d+"))) {
                    super.replace(fb, offset, length, text, attrs);
                }
            }
        });

        panel.add(new JLabel("Location:"));
        txtLocation = new JTextField();
        panel.add(txtLocation);

        defaultBorder = txtName.getBorder();

        // --- Catering Sections ---
        panel.add(new JLabel("--- قسم العيش (Rice) ---"));
        panel.add(new JLabel(""));

        panel.add(new JLabel("نوع العيش (Rice Dish):"));
        riceDishCombo = new JComboBox<>(CateringOrderCalculator.RICE_DISH_TYPES);
        panel.add(riceDishCombo);

        panel.add(new JLabel("كمية العيش (من / Mun):"));
        riceMunSpinner = new JSpinner(new SpinnerNumberModel(1, 1, 100, 1));
        panel.add(riceMunSpinner);

        // --- Unified Protein Section ---
        panel.add(new JLabel("--- قسم البروتين (Protein Section) ---"));
        panel.add(new JLabel(""));

        panel.add(new JLabel("اختر نوع البروتين:"));
        JPanel radioPanel = new JPanel(new FlowLayout(FlowLayout.LEFT, 5, 0));
        rbBeef = new JRadioButton("بقر", true);
        rbChicken = new JRadioButton("دجاج");
        rbFish = new JRadioButton("سمك");
        rbMutton = new JRadioButton("ضأن");
        proteinGroup = new ButtonGroup();
        proteinGroup.add(rbBeef);
        proteinGroup.add(rbChicken);
        proteinGroup.add(rbFish);
        proteinGroup.add(rbMutton);
        radioPanel.add(rbBeef);
        radioPanel.add(rbChicken);
        radioPanel.add(rbFish);
        radioPanel.add(rbMutton);
        panel.add(radioPanel);

        lblProteinDish = new JLabel("نوع طبق البقر (Beef Dish):");
        panel.add(lblProteinDish);
        proteinDishCombo = new JComboBox<>(CateringOrderCalculator.MEAT_DISH_TYPES);
        panel.add(proteinDishCombo);

        lblProteinQty = new JLabel("كمية البقر (من / Mun):");
        panel.add(lblProteinQty);

        proteinMunSpinner = new JSpinner(new SpinnerNumberModel(0, 0, 100, 1));
        panel.add(proteinMunSpinner);

        panel.add(new JLabel("إحضار الزبون (Customer Provided):"));
        proteinCustomerProvidedCheck = new JCheckBox("Customer Provided");
        panel.add(proteinCustomerProvidedCheck);

        // Radio button listeners (Stable layout without shifting)
        ActionListener proteinToggleListener = e -> {
            if (rbBeef.isSelected()) {
                lblProteinDish.setText("نوع طبق البقر (Beef Dish):");
                proteinDishCombo.setModel(new DefaultComboBoxModel<>(CateringOrderCalculator.MEAT_DISH_TYPES));
                lblProteinQty.setText("كمية البقر (من / Mun):");
                proteinCustomerProvidedCheck.setEnabled(true);
            } else if (rbChicken.isSelected()) {
                lblProteinDish.setText("نوع طبق الدجاج (Chicken Dish):");
                proteinDishCombo.setModel(new DefaultComboBoxModel<>(CateringOrderCalculator.CHICKEN_DISH_TYPES));
                lblProteinQty.setText("كمية الدجاج (من / Mun):");
                proteinCustomerProvidedCheck.setEnabled(true);
            } else if (rbFish.isSelected()) {
                lblProteinDish.setText("نوع طبق السمك (Fish Dish):");
                proteinDishCombo.setModel(new DefaultComboBoxModel<>(new String[]{"سمك مشوي"}));
                lblProteinQty.setText("كمية السمك (من / Mun):");
                proteinCustomerProvidedCheck.setEnabled(false);
                proteinCustomerProvidedCheck.setSelected(false);
            } else if (rbMutton.isSelected()) {
                lblProteinDish.setText("حجم الضأن (Mutton Size):");
                proteinDishCombo.setModel(new DefaultComboBoxModel<>(CateringOrderCalculator.MUTTON_SIZES));
                lblProteinQty.setText("عدد الذبائح (Quantity):");
                proteinCustomerProvidedCheck.setEnabled(true);
            }

            calculateCateringAndDue();
        };

        rbBeef.addActionListener(proteinToggleListener);
        rbChicken.addActionListener(proteinToggleListener);
        rbFish.addActionListener(proteinToggleListener);
        rbMutton.addActionListener(proteinToggleListener);
        // ---------------------------------

        // Order Details
        panel.add(new JLabel("Order Details Notes:"));
        txtOrder = new JTextField();
        panel.add(txtOrder);

        panel.add(new JLabel("Order Type:"));
        cmbOrderType = new JComboBox<>(new String[]{"Pickup", "Delivery"});
        panel.add(cmbOrderType);

        panel.add(new JLabel("Order Date:"));
        dateChooser = new JDateChooser();
        dateChooser.setDate(new Date());
        dateChooser.setDateFormatString("EEEE, dd-MM-yyyy");
        panel.add(dateChooser);

        panel.add(new JLabel("Time (12-Hour):"));
        JPanel timePanel = new JPanel(new FlowLayout(FlowLayout.LEFT, 0, 0));
        txtTime = new JTextField(10);
        cmbAmPm = new JComboBox<>(new String[]{"AM", "PM"});
        timePanel.add(txtTime);
        timePanel.add(Box.createHorizontalStrut(5));
        timePanel.add(cmbAmPm);
        panel.add(timePanel);

        // Payment Info
        panel.add(new JLabel("Total Price (OMR):"));
        txtPrice = new JTextField();
        panel.add(txtPrice);

        panel.add(new JLabel("Advance (OMR):"));
        txtAdvance = new JTextField();
        panel.add(txtAdvance);

        panel.add(new JLabel("Due Amount (OMR):"));
        txtDue = new JTextField();
        txtDue.setEditable(false);
        txtDue.setFocusable(false);
        panel.add(txtDue);

        panel.add(new JLabel("Payment Method:"));
        cmbPaymentMethod = new JComboBox<>(new String[]{"None", "Cash", "Card", "Online"});
        panel.add(cmbPaymentMethod);

        panel.add(new JLabel("Payment Status:"));
        cmbPaymentStatus = new JComboBox<>(new String[]{"Pending", "Partially Paid", "Paid"});
        panel.add(cmbPaymentStatus);

        // Buttons
        btnCalculate = new JButton("Recalculate");
        btnSave = new JButton("Save Order");
        btnClear = new JButton("Clear");
        btnBack = new JButton("Back");

        panel.add(btnCalculate);
        panel.add(btnSave);
        panel.add(btnClear);
        panel.add(btnBack);

        add(new JScrollPane(panel));
        getRootPane().setDefaultButton(btnSave);

        // Listeners for auto-calculation
        riceDishCombo.addActionListener(e -> calculateCateringAndDue());
        riceMunSpinner.addChangeListener(e -> calculateCateringAndDue());
        proteinDishCombo.addActionListener(e -> calculateCateringAndDue());
        proteinMunSpinner.addChangeListener(e -> calculateCateringAndDue());
        proteinCustomerProvidedCheck.addActionListener(e -> calculateCateringAndDue());

        DocumentListener autoCalcListener = new DocumentListener() {
            public void insertUpdate(DocumentEvent e) { calculateDue(); }
            public void removeUpdate(DocumentEvent e) { calculateDue(); }
            public void changedUpdate(DocumentEvent e) { calculateDue(); }
        };
        txtPrice.getDocument().addDocumentListener(autoCalcListener);
        txtAdvance.getDocument().addDocumentListener(autoCalcListener);

        btnCalculate.addActionListener(e -> calculateCateringAndDue());
        btnSave.addActionListener(e -> saveOrder());
        btnClear.addActionListener(e -> clearFields());
        btnBack.addActionListener(e -> confirmExitOrBack());

        calculateCateringAndDue();
        setVisible(true);
    }

    private void calculateCateringAndDue() {
        try {
            String selectedRice = riceDishCombo.getSelectedItem().toString();
            int riceMun = (int) riceMunSpinner.getValue();
            double riceTotal = orderCalculator.calculateRiceTotal(selectedRice, riceMun);

            double proteinTotal = 0.0;
            int proteinQty = (int) proteinMunSpinner.getValue();
            boolean isCustomerProvided = proteinCustomerProvidedCheck.isSelected();

            if (rbBeef.isSelected()) {
                proteinTotal = orderCalculator.calculateBeefTotal(proteinQty, isCustomerProvided);
            } else if (rbChicken.isSelected()) {
                proteinTotal = orderCalculator.calculateChickenTotal(proteinQty, isCustomerProvided);
            } else if (rbFish.isSelected()) {
                proteinTotal = orderCalculator.calculateFishTotal(proteinQty, isCustomerProvided);
            } else if (rbMutton.isSelected()) {
                String muttonSize = proteinDishCombo.getSelectedItem() != null ? proteinDishCombo.getSelectedItem().toString() : "";
                proteinTotal = orderCalculator.calculateMuttonTotal(muttonSize, proteinQty, isCustomerProvided);
            }

            double grandTotal = riceTotal + proteinTotal;
            txtPrice.setText(String.format(Locale.US, "%.3f", grandTotal));
        } catch (Exception ex) {
            // fallback
        }
        calculateDue();
    }

    private void resetFieldBorders() {
        txtName.setBorder(defaultBorder);
        txtPhone.setBorder(defaultBorder);
        txtLocation.setBorder(defaultBorder);
        txtOrder.setBorder(defaultBorder);
        txtTime.setBorder(defaultBorder);
        txtPrice.setBorder(defaultBorder);
        txtAdvance.setBorder(defaultBorder);
    }

    private void calculateDue() {
        try {
            String priceText = txtPrice.getText().trim();
            String advanceText = txtAdvance.getText().trim();

            double price = priceText.isEmpty() ? 0.0 : Double.parseDouble(priceText);
            double advance = advanceText.isEmpty() ? 0.0 : Double.parseDouble(advanceText);

            double due = Math.max(0, price - advance);
            txtDue.setText(String.format(Locale.US, "%.3f", due));

            if (price > 0) {
                if (advance >= price) {
                    cmbPaymentStatus.setSelectedItem("Paid");
                } else if (advance > 0) {
                    cmbPaymentStatus.setSelectedItem("Partially Paid");
                } else {
                    cmbPaymentStatus.setSelectedItem("Pending");
                }
            } else {
                cmbPaymentStatus.setSelectedItem("Pending");
            }
        } catch (NumberFormatException e) {
            txtDue.setText("Invalid input");
        }
    }

    private void confirmExitOrBack() {
        Object[] options = {"حفظ (Save)", "متابعة (Continue)", "خروج بدون حفظ (Exit without saving)"};
        int choice = JOptionPane.showOptionDialog(
                this,
                "هل تريد حفظ الطلب، العودة للمتابعة، أم الخروج؟",
                "تأكيد الخروج",
                JOptionPane.YES_NO_CANCEL_OPTION,
                JOptionPane.QUESTION_MESSAGE,
                null,
                options,
                options[1]
        );
        if (choice == 0) saveOrder();
        else if (choice == 2 || choice == JOptionPane.CLOSED_OPTION) dispose();
    }

    private void saveOrder() {
        resetFieldBorders();
        List<String> errors = new ArrayList<>();
        JTextField firstErrorField = null;

        String name = txtName.getText().trim();
        if (name.isEmpty()) {
            errors.add("اسم العميل مفقود");
            txtName.setBorder(errorBorder);
            if (firstErrorField == null) firstErrorField = txtName;
        }

        String phone = txtPhone.getText().trim();
        if (phone.length() != 8 || !phone.matches("^[97]\\d{7}$")) {
            errors.add("رقم الهاتف يجب أن يتكون من 8 أرقام ويبدأ بـ 9 أو 7");
            txtPhone.setBorder(errorBorder);
            if (firstErrorField == null) firstErrorField = txtPhone;
        }

        String priceStr = txtPrice.getText().trim();
        double priceVal = 0.0;
        try {
            priceVal = Double.parseDouble(priceStr);
            if (priceVal <= 0) {
                errors.add("يجب أن يحتوي الطلب على صنف واحد على الأقل (عيش أو بروتين)");
                txtPrice.setBorder(errorBorder);
                if (firstErrorField == null) firstErrorField = txtPrice;
            }
        } catch (NumberFormatException e) {
            errors.add("السعر الإجمالي غير صالح");
        }

        if (!errors.isEmpty()) {
            StringBuilder message = new StringBuilder("يرجى تصحيح الأخطاء:\n\n");
            for (String err : errors) message.append("• ").append(err).append("\n");
            JOptionPane.showMessageDialog(this, message.toString(), "خطأ", JOptionPane.WARNING_MESSAGE);
            if (firstErrorField != null) firstErrorField.requestFocus();
            return;
        }

        try {
            Date selectedDate = dateChooser.getDate() != null ? dateChooser.getDate() : new Date();
            String orderDate = new SimpleDateFormat("EEEE, dd-MM-yyyy", new Locale("ar")).format(selectedDate);

            Customer customer = new Customer(name, phone, txtLocation.getText().trim());
            String orderType = cmbOrderType.getSelectedItem().toString();
            String paymentStatus = cmbPaymentStatus.getSelectedItem().toString();
            String fullTime = txtTime.getText().trim() + " " + cmbAmPm.getSelectedItem().toString();
            int billNumber = OrderManager.generateOrderNumber();
            double advanceVal = txtAdvance.getText().trim().isEmpty() ? 0.0 : Double.parseDouble(txtAdvance.getText().trim());

            Order order = new Order(
                    customer, billNumber, txtOrder.getText().trim(), orderDate, fullTime,
                    priceVal, advanceVal, cmbPaymentMethod.getSelectedItem().toString(), paymentStatus, orderType
            );

            OrderManager.addOrder(order);
            JOptionPane.showMessageDialog(this, "تم حفظ الطلب بنجاح! رقم الطلب: " + billNumber);
            clearFields();
        } catch (Exception e) {
            JOptionPane.showMessageDialog(this, "خطأ أثناء الحفظ: " + e.getMessage(), "خطأ", JOptionPane.ERROR_MESSAGE);
        }
    }

    private void clearFields() {
        resetFieldBorders();
        txtName.setText("");
        txtPhone.setText("");
        txtLocation.setText("");
        txtOrder.setText("");
        txtTime.setText("");
        txtPrice.setText("");
        txtAdvance.setText("");
        txtDue.setText("");
        riceDishCombo.setSelectedIndex(0);
        riceMunSpinner.setValue(1);
        proteinMunSpinner.setValue(0);
        rbBeef.setSelected(true);
        lblProteinDish.setText("نوع طبق البقر (Beef Dish):");
        proteinDishCombo.setModel(new DefaultComboBoxModel<>(CateringOrderCalculator.MEAT_DISH_TYPES));
        lblProteinQty.setText("كمية البقر (من / Mun):");
        proteinCustomerProvidedCheck.setSelected(false);
        proteinCustomerProvidedCheck.setEnabled(true);
        dateChooser.setDate(new Date());
        calculateCateringAndDue();
    }
}
